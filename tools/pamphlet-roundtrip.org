* Overview

A *byte-reversible* pamphlet <-> org pair, the basis for migrating the
Axiom sources to org as the single source of truth: we must be able to
rebuild the exact =.pamphlet= (hence the exact PDF) from the =.org=.

The design that makes the round trip provable by a mere =cmp=:

- Transform ONLY the chunk scaffolding -- =\begin{chunk}{name}= <->
  =#+NAME: name= + =#+begin_src LANG :noweb yes=, =\end{chunk}= <->
  =#+end_src=, =\getchunk{name}= <-> =<<name>>=.
- Emit *every other line verbatim* (prose, preamble, headings, verbatim,
  bibliography, structural markers).  Identity is trivially reversible.
- No added/removed blank lines.

So =to-pamphlet(to-org(P)) == P= byte-for-byte for the clean majority;
edge cases (malformed markers, trailing text) are handled as we find them.

* Forward: pamphlet -> org

#+BEGIN_SRC elisp
(defun pamphlet-roundtrip-to-org (text filename)
  "Convert pamphlet TEXT to reversible org.  FILENAME picks the src lang."
  ;; Keep ALL split fields (including the trailing "" from a final \n, and any
  ;; empty lines) and join without appending -- this preserves the exact
  ;; trailing-newline state, so a file with no final newline round-trips too.
  (let ((out '()) (in-chunk nil) (in-verbatim nil) (lines (split-string text "\n")))
    (dolist (line lines)
      (let ((trimmed (string-trim line)))
        (cond
         (in-chunk
          (cond
           ;; \end{chunk}EXTRA at column 0 -- carry EXTRA (trailing space, a
           ;; malformed "p" or "{*}") on the #+end_src line so it reverses exactly
           ((string-prefix-p "\\end{chunk}" line)
            (push (concat pamphlet2org--src-prefix "end_src"
                          (substring line (length "\\end{chunk}")))
                  out)
            (setq in-chunk nil))
           ((string-match "\\`\\\\getchunk{\\([^}]*\\)}\\'" line)
            (push (format "<<%s>>" (match-string 1 line)) out))
           (t (push line out))))
         (in-verbatim
          (push line out)
          (when (string-suffix-p "\\end{verbatim}" trimmed)
            (setq in-verbatim nil)))
         ;; \begin{chunk}{NAME}EXTRA at column 0 -- NAME up to the first }, EXTRA
         ;; (e.g. a stray }) appended to #+begin_src for exact reversal
         ((string-match "\\`\\\\begin{chunk}{\\([^}]*\\)}\\(.*\\)\\'" line)
          (let ((name (match-string 1 line)) (extra (match-string 2 line)))
            (push (format "%sNAME: %s" pamphlet2org--src-prefix name) out)
            (push (format "%sbegin_src %s :noweb yes%s"
                          pamphlet2org--src-prefix
                          (pamphlet2org--detect-lang name filename) extra)
                  out)
            (setq in-chunk t)))
         ((string= trimmed "\\begin{verbatim}")
          (push line out)
          (setq in-verbatim t))
         ;; getchunk/<<>> are only converted INSIDE chunks (where org-babel
         ;; noweb actually expands them).  In prose, both \getchunk{x} and a
         ;; literal <<x>> stay verbatim, so a literal <<x>> (e.g. bookvol11)
         ;; is never mistaken for a reference on the way back.
         (t (push line out)))))
    (string-join (nreverse out) "\n")))
#+END_SRC

* Inverse: org -> pamphlet

#+BEGIN_SRC elisp
(defun pamphlet-roundtrip-to-pamphlet (text)
  "Reverse `pamphlet-roundtrip-to-org': org TEXT back to pamphlet."
  (let ((out '()) (lines (split-string text "\n")) (pfx pamphlet2org--src-prefix)
        (in-src nil))
    (while lines
      (let ((line (pop lines)))
        (cond
         ((string-prefix-p (concat pfx "NAME: ") line)
          (let ((name (substring line (length (concat pfx "NAME: ")))))
            (if (and lines (string-prefix-p (concat pfx "begin_src ") (car lines)))
                (let* ((bs (pop lines))      ; #+begin_src LANG :noweb yesEXTRA
                       (extra (if (string-match ":noweb yes\\(.*\\)\\'" bs)
                                  (match-string 1 bs) "")))
                  (push (format "\\begin{chunk}{%s}%s" name extra) out)
                  (setq in-src t))
              (push line out))))             ; lone #+NAME in prose: keep raw
         ((and in-src (string-prefix-p (concat pfx "end_src") line))
          (push (concat "\\end{chunk}"
                        (substring line (length (concat pfx "end_src"))))
                out)
          (setq in-src nil))
         ;; <<x>> reverts to \getchunk{x} only inside a chunk (src block);
         ;; a literal <<x>> in prose stays as-is
         ((and in-src (string-match "\\`<<\\([^>]*\\)>>\\'" line))
          (push (format "\\getchunk{%s}" (match-string 1 line)) out))
         (t (push line out)))))
    (string-join (nreverse out) "\n")))
#+END_SRC

* Round-trip test

#+BEGIN_SRC elisp
(defun pamphlet-roundtrip-test-file (pamphlet)
  "Return t if PAMPHLET survives pamphlet->org->pamphlet byte-for-byte."
  (let* ((orig (with-temp-buffer
                 (set-buffer-multibyte nil)
                 (insert-file-contents-literally pamphlet)
                 (buffer-string)))
         (org (pamphlet-roundtrip-to-org orig (file-name-nondirectory pamphlet)))
         (back (pamphlet-roundtrip-to-pamphlet org)))
    (string= orig back)))
#+END_SRC

* Corpus round-trip E2E

Walk every =.pamphlet= under ROOT and assert each survives the round trip
byte-for-byte.  This is the migration's go/no-go gate for losslessness:
if a file rebuilds exactly, the =.org= provably holds all of it (and the
rebuilt =.pamphlet= compiles to the identical PDF -- same input, same
output).  =AX=/path emacs --batch ... --eval '(pamphlet-roundtrip-test-all)'.

#+BEGIN_SRC elisp
(require 'cl-lib)
(require 'seq)

(defun pamphlet-roundtrip-test-all (&optional root)
  "Round-trip every .pamphlet under ROOT; report an OK/DIFFER tally.
Returns (:ok N :differ N :failures (RELPATH ...))."
  (let* ((root (or root (getenv "AX") default-directory))
         (files (seq-filter (lambda (x) (not (string-match-p "/\\.git/" x)))
                            (directory-files-recursively root "\\.pamphlet\\'")))
         (ok 0) (bad 0) (fails '()))
    (dolist (f files)
      (condition-case _
          (if (pamphlet-roundtrip-test-file f)
              (cl-incf ok)
            (cl-incf bad) (push (file-relative-name f root) fails))
        (error (cl-incf bad) (push (concat (file-name-nondirectory f) " (error)")
                                   fails))))
    (message "round-trip: %d OK / %d DIFFER (of %d)" ok bad (length files))
    (dolist (x (nreverse fails)) (message "  FAIL %s" x))
    (list :ok ok :differ bad :failures fails)))
#+END_SRC

* Migration + build helpers

The migration makes =X.pamphlet.org= the source and treats =X.pamphlet=
as a regenerated build artifact.  =migrate-all= writes the org files once;
=regen-all= rebuilds the exact pamphlets from them at build time (so the
existing =tanglec= / =make= rules keep working byte-for-byte).

#+BEGIN_SRC elisp
(defun pamphlet-roundtrip-convert-file (pamphlet)
  "Write PAMPHLET's reversible org beside it as PAMPHLET\".org\"."
  (let ((orig (with-temp-buffer (set-buffer-multibyte nil)
                (insert-file-contents-literally pamphlet) (buffer-string)))
        (coding-system-for-write 'binary))
    (with-temp-file (concat pamphlet ".org")
      (insert (pamphlet-roundtrip-to-org orig (file-name-nondirectory pamphlet))))
    (concat pamphlet ".org")))

(defun pamphlet-roundtrip-regen-file (orgfile)
  "Rebuild the .pamphlet from ORGFILE (a *.pamphlet.org); return its path."
  (let ((pamphlet (string-remove-suffix ".org" orgfile))
        (org (with-temp-buffer (set-buffer-multibyte nil)
               (insert-file-contents-literally orgfile) (buffer-string)))
        (coding-system-for-write 'binary))
    (with-temp-file pamphlet
      (insert (pamphlet-roundtrip-to-pamphlet org)))
    pamphlet))

(defun pamphlet-roundtrip-migrate-all (&optional root)
  "Convert every .pamphlet under ROOT to a sibling .pamphlet.org."
  (let ((root (or root (getenv "AX") default-directory)) (n 0))
    (dolist (f (seq-filter (lambda (x) (not (string-match-p "/\\.git/" x)))
                           (directory-files-recursively root "\\.pamphlet\\'")))
      (pamphlet-roundtrip-convert-file f) (cl-incf n))
    (message "migrated %d pamphlets -> .pamphlet.org" n) n))

(defun pamphlet-roundtrip-regen-all (&optional root)
  "Rebuild every .pamphlet from its .pamphlet.org under ROOT (build step)."
  (let ((root (or root (getenv "AX") default-directory)) (n 0))
    (dolist (f (directory-files-recursively root "\\.pamphlet\\.org\\'"))
      (unless (string-match-p "/\\.git/" f)
        (pamphlet-roundtrip-regen-file f) (cl-incf n)))
    (message "regenerated %d pamphlets from .org" n) n))
#+END_SRC
