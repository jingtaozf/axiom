# -*- mode: org; -*-
#+TITLE: test-pamphlet2org -- E2E verification for pamphlet to org conversion
#+AUTHOR: Jingtao Xu
#+PROPERTY: header-args :results silent

* Overview

End-to-end test: convert each =.pamphlet= file to =.org=, tangle both,
compare output.  Reports pass/fail per file.

Usage:

#+begin_example
emacs --batch -l tools/pamphlet2org.org -l tools/test-pamphlet2org.el \
  --eval '(test-pamphlet2org-run "src/interp")'
#+end_example

* Implementation

** Dependencies

#+BEGIN_SRC emacs-lisp
(require 'pamphlet2org)
#+END_SRC

** Chunk map (named-chunk pamphlets)

A few pamphlets (=bookvol5/9/10.5=) have no =*= root chunk — the real
build extracts a named chunk (Interpreter / Compiler / Numerics).
Derive the map from the directory's =Makefile= instead of hard-coding
names, so new =(tangle …)= targets are picked up automatically.

#+BEGIN_SRC emacs-lisp
(defun test-pamphlet2org--chunk-map (dir)
  "Build a basename->chunk alist from DIR/Makefile.
Scan for (tangle \"X.pamphlet\" \"Chunk\" ...) forms and keep only the
ones whose chunk is not the default \"*\"; pamphlets not listed there
default to \"*\" anyway."
  (let ((makefile (expand-file-name "Makefile" dir))
        (map '()))
    (when (file-readable-p makefile)
      (with-temp-buffer
        (insert-file-contents makefile)
        (goto-char (point-min))
        (while (re-search-forward
                "(tangle[ \t]+\"[^\"]*?\\([^\"/]+\\.pamphlet\\)\"[ \t]+\"\\([^\"]+\\)\""
                nil t)
          (let ((file (match-string 1))
                (ch (match-string 2)))
            (unless (or (string= ch "*") (assoc file map))
              (push (cons file ch) map))))))
    (nreverse map)))
#+END_SRC

** Test runner

#+BEGIN_SRC emacs-lisp
(defun test-pamphlet2org-run (dir &optional project-root)
  "Run E2E test on all .pamphlet files under DIR.
PROJECT-ROOT defaults to parent of DIR.
Reports pass/fail per file with diff summary."
  (let* ((root (or project-root
                   (expand-file-name ".." (directory-file-name dir))))
         (full-dir (expand-file-name dir root))
         (tmp-dir (make-temp-file "pamphlet2org-test-" t))
         (tanglec (expand-file-name "books/tanglec" root))
         (files (directory-files full-dir t "\\.pamphlet$"))
         (chunk-map (test-pamphlet2org--chunk-map full-dir))
         (pass 0) (fail 0) (skip 0)
         (failures '()))
    ;; Build tanglec if needed
    (unless (file-executable-p tanglec)
      (message "Building tanglec...")
      (shell-command (format "cd %s && gcc -o books/tanglec books/tanglec.c"
                             (shell-quote-argument root))))
    ;; Run tests
    (message "Testing %d pamphlet files in %s..." (length files) dir)
    (dolist (pamphlet files)
      (let* ((base (file-name-nondirectory pamphlet))
             (chunk (cdr (assoc base chunk-map)))
             (name (file-name-sans-extension base))
             (ref-file (expand-file-name (concat name ".ref") tmp-dir))
             (org-file (expand-file-name (concat name ".org") tmp-dir))
             (test-file (expand-file-name (concat name ".test") tmp-dir)))
        (condition-case err
            (progn
              ;; Step 1: Tangle pamphlet with tanglec
              (shell-command (format "%s %s %s > %s"
                                     (shell-quote-argument tanglec)
                                     (shell-quote-argument pamphlet)
                                     (if chunk (shell-quote-argument chunk) "")
                                     (shell-quote-argument ref-file)))
              ;; Step 2: Convert pamphlet to org
              (pamphlet2org-convert-file pamphlet org-file)
              ;; Step 3: Tangle org
              (let ((tangled (pamphlet2org--tangle org-file chunk)))
                ;; tangled holds raw eight-bit bytes from the binary-read
                ;; pamphlet; write them verbatim, else a non-UTF-8 byte makes
                ;; batch Emacs block on a "Select coding system" prompt.
                (let ((coding-system-for-write 'binary))
                  (with-temp-file test-file (insert tangled))))
              ;; Step 4: Compare
              (let ((diff-result (shell-command-to-string
                                  (format "diff %s %s"
                                          (shell-quote-argument ref-file)
                                          (shell-quote-argument test-file)))))
                (if (string= diff-result "")
                    (progn
                      (setq pass (1+ pass))
                      (message "  PASS: %s" base))
                  (setq fail (1+ fail))
                  (let ((first-line (car (split-string diff-result "\n"))))
                    (push (cons base first-line) failures)
                    (message "  FAIL: %s -- %s" base first-line)))))
          (error
           (setq skip (1+ skip))
           (push (cons base (error-message-string err)) failures)
           (message "  SKIP: %s -- %s" base (error-message-string err))))))
    ;; Report
    (message "\n=== Results ===")
    (message "Total: %d | Pass: %d | Fail: %d | Skip: %d"
             (+ pass fail skip) pass fail skip)
    (when failures
      (message "\nFailures:")
      (dolist (f (nreverse failures))
        (message "  %s: %s" (car f) (cdr f))))
    ;; Cleanup
    (delete-directory tmp-dir t)
    (list :pass pass :fail fail :skip skip :failures failures)))
#+END_SRC

** Quick test (single file)

#+BEGIN_SRC emacs-lisp
(defun test-pamphlet2org-file (pamphlet-file)
  "Test a single pamphlet file. Returns t if identical."
  (let* ((tmp-dir (make-temp-file "p2o-single-" t))
         (tanglec (expand-file-name
                   "books/tanglec"
                   (locate-dominating-file pamphlet-file "books")))
         (name (file-name-sans-extension
                (file-name-nondirectory pamphlet-file)))
         (ref-file (expand-file-name (concat name ".ref") tmp-dir))
         (org-file (expand-file-name (concat name ".org") tmp-dir))
         (test-file (expand-file-name (concat name ".test") tmp-dir))
         result)
    (shell-command (format "%s %s > %s"
                           (shell-quote-argument tanglec)
                           (shell-quote-argument pamphlet-file)
                           (shell-quote-argument ref-file)))
    (pamphlet2org-convert-file pamphlet-file org-file)
    ;; binary write: raw eight-bit bytes must not trigger a batch coding prompt.
    (let ((coding-system-for-write 'binary))
      (with-temp-file test-file
        (insert (pamphlet2org--tangle org-file))))
    (setq result (= 0 (shell-command
                        (format "diff %s %s"
                                (shell-quote-argument ref-file)
                                (shell-quote-argument test-file)))))
    (delete-directory tmp-dir t)
    (message "%s: %s" (file-name-nondirectory pamphlet-file)
             (if result "PASS" "FAIL"))
    result))
#+END_SRC

** Per-chunk test (src/algebra book volumes)

The =src/algebra= volumes (=bookvol10.2..10.5=) have no =*= root chunk —
they are huge literate books whose thousands of named chunks are tangled
per domain (=category ATADDVA …=, =package RECOP …=, …).  The whole-file
runner above does not apply; instead compare *every named chunk* against
=tanglec=.  =STRIDE= > 1 tests every STRIDE-th chunk (sampling) for the
biggest files.

#+BEGIN_SRC emacs-lisp
(defun test-pamphlet2org-chunks (pamphlet-file &optional project-root stride)
  "Per-chunk E2E for PAMPHLET-FILE: tangle every named chunk via tanglec
and via pamphlet2org and compare.  PROJECT-ROOT defaults to the nearest
ancestor containing =books=.  STRIDE (default 1) tests every STRIDE-th
chunk.  Returns (:total N :pass N :fail N :failures NAMES)."
  (let* ((root (or project-root (locate-dominating-file pamphlet-file "books")))
         (tanglec (expand-file-name "books/tanglec" root))
         (tmp-dir (make-temp-file "p2o-chunks-" t))
         (org (expand-file-name "x.org" tmp-dir))
         (stride (or stride 1))
         (pass 0) (fail 0) (failures '())
         (visiting (make-hash-table :test #'equal)))
    (unless (file-executable-p tanglec)
      (shell-command (format "cd %s && gcc -o books/tanglec books/tanglec.c"
                             (shell-quote-argument root))))
    (pamphlet2org-convert-file pamphlet-file org)
    (let ((blocks (pamphlet2org--collect-blocks org))
          (names '()) (i -1))
      (maphash (lambda (k _v) (push k names)) blocks)
      (setq names (nreverse names))
      (dolist (nm names)
        (when (= 0 (mod (setq i (1+ i)) stride))
          (let ((ref (shell-command-to-string
                      (format "%s %s %s" (shell-quote-argument tanglec)
                              (shell-quote-argument pamphlet-file)
                              (shell-quote-argument nm))))
                (out (generate-new-buffer " *p2o-chunk*")))
            (clrhash visiting)
            (pamphlet2org--emit (gethash nm blocks) blocks "" visiting out)
            (let ((got (with-current-buffer out (buffer-string))))
              (kill-buffer out)
              (if (string= ref got) (setq pass (1+ pass))
                (setq fail (1+ fail))
                (push nm failures))))))
      (message "%s: %d chunks | pass=%d fail=%d"
               (file-name-nondirectory pamphlet-file) (length names) pass fail)
      (dolist (f (nreverse failures)) (message "  FAIL: %s" f))
      (delete-directory tmp-dir t)
      (list :total (length names) :pass pass :fail fail :failures failures))))
#+END_SRC
