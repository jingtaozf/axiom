#+TITLE: test-pamphlet2org -- E2E verification for pamphlet to org conversion
#+AUTHOR: Jingtao Xu
#+PROPERTY: header-args :results silent

* Overview

End-to-end test: convert each =.pamphlet= file to =.org=, tangle both,
compare output.  Reports pass/fail per file.

Usage:

#+begin_example
emacs --batch -l tools/pamphlet2org.org -l tools/test-pamphlet2org.el \
  --eval '(test-pamphlet2org-run "src/interp")'
#+end_example

* Implementation

** Dependencies

#+BEGIN_SRC emacs-lisp
(require 'pamphlet2org)
#+END_SRC

** Test runner

#+BEGIN_SRC emacs-lisp
(defun test-pamphlet2org-run (dir &optional project-root)
  "Run E2E test on all .pamphlet files under DIR.
PROJECT-ROOT defaults to parent of DIR.
Reports pass/fail per file with diff summary."
  (let* ((root (or project-root
                   (expand-file-name ".." (directory-file-name dir))))
         (full-dir (expand-file-name dir root))
         (tmp-dir (make-temp-file "pamphlet2org-test-" t))
         (tanglec (expand-file-name "books/tanglec" root))
         (files (directory-files full-dir t "\\.pamphlet$"))
         (pass 0) (fail 0) (skip 0)
         (failures '()))
    ;; Build tanglec if needed
    (unless (file-executable-p tanglec)
      (message "Building tanglec...")
      (shell-command (format "cd %s && gcc -o books/tanglec books/tanglec.c"
                             (shell-quote-argument root))))
    ;; Run tests
    (message "Testing %d pamphlet files in %s..." (length files) dir)
    (dolist (pamphlet files)
      (let* ((base (file-name-nondirectory pamphlet))
             (name (file-name-sans-extension base))
             (ref-file (expand-file-name (concat name ".ref") tmp-dir))
             (org-file (expand-file-name (concat name ".org") tmp-dir))
             (test-file (expand-file-name (concat name ".test") tmp-dir)))
        (condition-case err
            (progn
              ;; Step 1: Tangle pamphlet with tanglec
              (shell-command (format "%s %s > %s"
                                     (shell-quote-argument tanglec)
                                     (shell-quote-argument pamphlet)
                                     (shell-quote-argument ref-file)))
              ;; Step 2: Convert pamphlet to org
              (pamphlet2org-convert-file pamphlet org-file)
              ;; Step 3: Tangle org
              (let ((tangled (pamphlet2org--tangle org-file)))
                (with-temp-file test-file (insert tangled)))
              ;; Step 4: Compare
              (let ((diff-result (shell-command-to-string
                                  (format "diff %s %s"
                                          (shell-quote-argument ref-file)
                                          (shell-quote-argument test-file)))))
                (if (string= diff-result "")
                    (progn
                      (setq pass (1+ pass))
                      (message "  PASS: %s" base))
                  (setq fail (1+ fail))
                  (let ((first-line (car (split-string diff-result "\n"))))
                    (push (cons base first-line) failures)
                    (message "  FAIL: %s -- %s" base first-line)))))
          (error
           (setq skip (1+ skip))
           (push (cons base (error-message-string err)) failures)
           (message "  SKIP: %s -- %s" base (error-message-string err))))))
    ;; Report
    (message "\n=== Results ===")
    (message "Total: %d | Pass: %d | Fail: %d | Skip: %d"
             (+ pass fail skip) pass fail skip)
    (when failures
      (message "\nFailures:")
      (dolist (f (nreverse failures))
        (message "  %s: %s" (car f) (cdr f))))
    ;; Cleanup
    (delete-directory tmp-dir t)
    (list :pass pass :fail fail :skip skip :failures failures)))
#+END_SRC

** Quick test (single file)

#+BEGIN_SRC emacs-lisp
(defun test-pamphlet2org-file (pamphlet-file)
  "Test a single pamphlet file. Returns t if identical."
  (let* ((tmp-dir (make-temp-file "p2o-single-" t))
         (tanglec (expand-file-name
                   "books/tanglec"
                   (locate-dominating-file pamphlet-file "books")))
         (name (file-name-sans-extension
                (file-name-nondirectory pamphlet-file)))
         (ref-file (expand-file-name (concat name ".ref") tmp-dir))
         (org-file (expand-file-name (concat name ".org") tmp-dir))
         (test-file (expand-file-name (concat name ".test") tmp-dir))
         result)
    (shell-command (format "%s %s > %s"
                           (shell-quote-argument tanglec)
                           (shell-quote-argument pamphlet-file)
                           (shell-quote-argument ref-file)))
    (pamphlet2org-convert-file pamphlet-file org-file)
    (with-temp-file test-file
      (insert (pamphlet2org--tangle org-file)))
    (setq result (= 0 (shell-command
                        (format "diff %s %s"
                                (shell-quote-argument ref-file)
                                (shell-quote-argument test-file)))))
    (delete-directory tmp-dir t)
    (message "%s: %s" (file-name-nondirectory pamphlet-file)
             (if result "PASS" "FAIL"))
    result))
#+END_SRC
