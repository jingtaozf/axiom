)abbrev package DONSIMP donSimple
donSimple(): Exports == Implementation where
  F   ==> Float
  SF  ==> Segment F
  EF  ==> Expression F
  SBF ==> SegmentBinding F

  Exports ==> with
    simple: (F->F,SF) -> F
    ++ simple(a,b) 
    ++
    ++X simple(x+->x^2,1..2)

    simple: (EF,SBF) -> EF
    ++ simple(a,b) 
    ++
    ++X simple(x^2,x=1..2)

  Implementation ==> add
    simple(func:F->F, sf:SF) ==
      a:F := lo(sf)
      b:F := hi(sf)
      func(b) - func(a)

    simple(func:EF, sbf:SBF) ==
      a:F := lo(segment(sbf))
      b:F := hi(segment(sbf))
      x:Symbol := variable(sbf)
      eval(func,kernel(x),b::EF) - eval(func,kernel(x),a::EF)
#+end_src
